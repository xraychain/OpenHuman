cmake_minimum_required(VERSION 3.16)
project(OpenHuman VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(OPENHUMAN_OPENMP "Parallelize diffusion with OpenMP" OFF)
option(OPENHUMAN_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
add_library(openhuman_options INTERFACE)
target_include_directories(openhuman_options INTERFACE src)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(openhuman_options INTERFACE -Wall -Wextra -Wpedantic -Wno-unknown-pragmas)
  if(OPENHUMAN_SANITIZERS)
    target_compile_options(openhuman_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(openhuman_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()
if(OPENHUMAN_OPENMP)
  find_package(OpenMP REQUIRED)
  target_link_libraries(openhuman_options INTERFACE OpenMP::OpenMP_CXX)
endif()
add_executable(openhuman src/main.cpp)
target_link_libraries(openhuman PRIVATE openhuman_options)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core.cpp)
  target_link_libraries(core_tests PRIVATE openhuman_options)
  add_test(NAME core COMMAND core_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py $<TARGET_FILE:openhuman>)
  set_tests_properties(core cli PROPERTIES TIMEOUT 60)
endif()
